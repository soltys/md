cmake_minimum_required(VERSION 4.0)

# Static CRT everywhere (our code and every fetched dependency) so the result
# is a single self-contained md.exe. Must be set before project().
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

project(md VERSION 0.2.0 LANGUAGES C CXX) # CXX only for HarfBuzz; app code is C

if(NOT MSVC)
    message(WARNING "md targets the MSVC toolchain. Other compilers are only "
                    "supported for quick local sanity builds.")
endif()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Whole-program optimisation for Release builds (applies to dependencies too).
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_MINSIZEREL ON)

# Keep the folder with md.exe clean: every other build output (libraries of
# the fetched dependencies, PDBs, import libraries, pixelart.exe, ...) goes to
# <build>/artifacts[/<Config>]. Only md.exe is redirected back, see below.
# Multi-config generators (Visual Studio) append the configuration name, so
# the result is build/vs2026/Release/md.exe and build/vs2026/artifacts/Release.
set(MD_ARTIFACTS_DIR "${CMAKE_BINARY_DIR}/artifacts")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${MD_ARTIFACTS_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${MD_ARTIFACTS_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${MD_ARTIFACTS_DIR}")
set(CMAKE_PDB_OUTPUT_DIRECTORY     "${MD_ARTIFACTS_DIR}")

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(Dependencies)
include(Fonts)
include(Grammars)

add_executable(md WIN32
    src/main.c
    src/app.h
    src/assets.c
    src/assets.h
    src/asset_ids.h
    src/fonts.c
    src/fonts.h
    src/document.c
    src/document.h
    src/view.c
    src/view.h
    src/theme.h
    src/nvim.c
    src/nvim.h
    src/msgpack.c
    src/msgpack.h
    src/editor.c
    src/editor.h
    src/cli.c
    src/cli.h
    src/clipboard.c
    src/clipboard.h
    src/emoji.c
    src/emoji.h

    src/grammar_pack.c
    src/grammar_pack.h
    src/highlight.c
    src/highlight.h
    src/regex.c
    src/regex.h
    src/timing.c
    src/timing.h
    ${MD_QUERIES_SOURCE}

# Neovim config for the editor pane, embedded like the fonts.
set(MD_NVIM_INIT "${CMAKE_CURRENT_SOURCE_DIR}/nvim/init.lua")
# App icon, generated by pixelart (soltys/pixelart) from art/sprites.txt and committed.
set(MD_APP_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/art/png/md-icon.ico")

# md.exe alone lands in <build>/<Config> (or <build> for single-config
# generators); its PDB and any import library stay in the artifacts folder.
set_target_properties(md PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

target_include_directories(md PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
# The version comes from project() above; md --version prints it and on
# Windows it is also in the file's properties (Details tab).
target_compile_definitions(md PRIVATE MD_VERSION="${PROJECT_VERSION}")

# Startup timing log (src/timing.h): every step from process start to the
# first full render, written to the debugger, the console and
# %TEMP%\md-timing.log. Always on in Debug; other configurations only with
# -DMD_TIMING_LOG=ON. When off, the logging compiles out completely.
option(MD_TIMING_LOG "Log startup timing in Release builds too (Debug always logs)" OFF)
target_compile_definitions(md PRIVATE
    $<IF:$<OR:$<CONFIG:Debug>,$<BOOL:${MD_TIMING_LOG}>>,MD_TIMING=1,MD_TIMING=0>)
if(MD_TIMING_LOG)
    message(STATUS "md startup timing log: ON in every configuration (MD_TIMING_LOG=ON)")
else()
    message(STATUS "md startup timing log: Debug only (Release: -DMD_TIMING_LOG=ON)")
endif()

target_link_libraries(md PRIVATE

    SDL3_image::SDL3_image-static
    md_tree_sitter

    md_ts_markdown_inline

if(WIN32)

    # they are reached through LockResource(), i.e. straight out of the mapped

    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/generated/assets.rc" PROPERTIES

    target_sources(md PRIVATE
        "${CMAKE_CURRENT_BINARY_DIR}/generated/assets.rc"
        src/md.manifest)

if(MSVC)
    target_compile_options(md PRIVATE /W4 /utf-8 /permissive- /Gy /Gw /Oi
        $<$<CONFIG:Release,MinSizeRel>:/GS->)
    target_compile_definitions(md PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_options(md PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO)
    # Windows loads every DLL in md.exe's import table (and their imports)
    # before main() runs. These are not needed to put the first page on
    # screen, or only later on: load them when first called instead. The
    # startup log lists the DLLs that are still loaded before main().
    # /IGNORE:4199: no warning for a DLL that this build does not import.
    target_link_options(md PRIVATE
        /DELAYLOAD:shell32.dll /DELAYLOAD:ole32.dll /DELAYLOAD:oleaut32.dll
        /DELAYLOAD:setupapi.dll /DELAYLOAD:cfgmgr32.dll /DELAYLOAD:version.dll
        /DELAYLOAD:winmm.dll /IGNORE:4199)
    target_link_libraries(md PRIVATE delayimp)
else()
    target_compile_options(md PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

# Install: md.exe is self-contained (fonts and the Neovim config are
# embedded), so installing is copying one file. On Windows the default
# location (a folder meant to be on PATH) is the first of:
#   1. the PATH_APPS environment variable,
#   2. %USERPROFILE%\Documents\PathApps, if that folder exists,
#   3. %USERPROFILE%\PathApps.
# An explicit -DCMAKE_INSTALL_PREFIX or --prefix always wins.
#   cmake --build --preset install
#   cmake --install build/vs2026 --config Release [--prefix <dir>]
if(WIN32)
    set(_md_prefix "")
    if(DEFINED ENV{PATH_APPS} AND NOT "$ENV{PATH_APPS}" STREQUAL "")
        file(TO_CMAKE_PATH "$ENV{PATH_APPS}" _md_prefix)
    elseif(DEFINED ENV{USERPROFILE})
        file(TO_CMAKE_PATH "$ENV{USERPROFILE}" _md_home)
        if(IS_DIRECTORY "${_md_home}/Documents/PathApps")
            set(_md_prefix "${_md_home}/Documents/PathApps")
        else()
            set(_md_prefix "${_md_home}/PathApps")
        endif()
    endif()

    set(_md_old_default "")
    if(DEFINED ENV{LOCALAPPDATA})
        file(TO_CMAKE_PATH "$ENV{LOCALAPPDATA}/Programs/md" _md_old_default)
    endif()
    # Use the default unless the prefix was chosen by hand: replace CMake's
    # stock default (C:/Program Files/md, which needs admin), a value this
    # rule picked on an earlier configure (so changing PATH_APPS takes effect
    # in an existing build tree), and the defaults of older md versions.
    if(_md_prefix AND (
       CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT OR
       CMAKE_INSTALL_PREFIX STREQUAL "${MD_AUTO_INSTALL_PREFIX}" OR
       CMAKE_INSTALL_PREFIX STREQUAL _md_old_default OR
       CMAKE_INSTALL_PREFIX MATCHES "/Documents/PathApps$" OR
       CMAKE_INSTALL_PREFIX MATCHES "/Program Files( \\(x86\\))?/md$"))
        set(CMAKE_INSTALL_PREFIX "${_md_prefix}" CACHE PATH "Install location" FORCE)
        set(MD_AUTO_INSTALL_PREFIX "${_md_prefix}" CACHE INTERNAL "")
    endif()
    message(STATUS "md.exe and mdv.exe install to ${CMAKE_INSTALL_PREFIX}")
    install(TARGETS md RUNTIME DESTINATION .)
    # The same program a second time, as mdv.exe (a short name for viewing).
    install(PROGRAMS $<TARGET_FILE:md> DESTINATION . RENAME mdv.exe)
else()
    install(TARGETS md RUNTIME DESTINATION bin)
    install(PROGRAMS $<TARGET_FILE:md> DESTINATION bin RENAME mdv)
endif()

# pixelart.exe (https://github.com/soltys/pixelart) turns the text sprite
# sheets in art/ into the PNG/ICO assets in art/png. md.exe does not depend
# on it: the images are committed, so a normal build never runs it.
option(MD_BUILD_TOOLS "Fetch and build pixelart.exe for the 'art' target" ON)
if(MD_BUILD_TOOLS AND WIN32)
    include(FetchContent)
    FetchContent_Declare(pixelart
        GIT_REPOSITORY https://github.com/soltys/pixelart.git
        GIT_TAG        3112b9b7bc5c6bc111caadbb2d8666808ffaf95e # main, 2026-11-13
        SYSTEM)
    FetchContent_MakeAvailable(pixelart)

    # `cmake --build --preset release --target art` regenerates art/png.
    file(GLOB MD_ART_SHEETS CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/art/*.txt")
    add_custom_target(art
        COMMAND pixelart -o "${PROJECT_SOURCE_DIR}/art/png" ${MD_ART_SHEETS}
        DEPENDS pixelart ${MD_ART_SHEETS}
        WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
        COMMENT "Generating PNG assets from art/*.txt"
        VERBATIM)
endif()
